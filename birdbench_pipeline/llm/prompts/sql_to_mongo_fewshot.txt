### Example 2
SQL:
SELECT T1.CustomerID
FROM customers T1
JOIN yearmonth T2 ON T1.CustomerID = T2.CustomerID
WHERE T1.Segment = 'LAM'
GROUP BY T1.CustomerID
ORDER BY SUM(T2.Consumption) ASC
LIMIT 1

MongoDB:
[
  {
    "$lookup": {
      "from": "yearmonth",
      "localField": "CustomerID",
      "foreignField": "CustomerID",
      "as": "yearmonth_data"
    }
  },
  { "$unwind": "$yearmonth_data" },
  { "$match": { "Segment": "LAM" } },
  {
    "$group": {
      "_id": "$CustomerID",
      "totalConsumption": {
        "$sum": "$yearmonth_data.Consumption"
      }
    }
  },
  { "$sort": { "totalConsumption": 1 } },
  { "$limit": 1 },
  { "$project": { "CustomerID": "$_id", "_id": 0 } }
]
